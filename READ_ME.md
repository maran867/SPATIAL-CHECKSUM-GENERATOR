OVERVIEW:
# The Automated Spatial Checksum Generator is a Python-based tool designed to monitor files and identify changes using cryptographic checksums.

# It generates SHA-256 hashes for files, stores the baseline checksum values in JSON format, and compares current files against the baseline to detect modifications.

FEATURES:

# Scans files inside a project directory.

# Calculates SHA-256 checksums using Python's hashlib module.

# Stores checksum values in JSON format.

# Detects newly added files.

# Identifies modified files.

# Detects deleted files.

# Helps maintain file integrity for geospatial datasets and other project files.

TECHNOLOGIES:

# Python

# pathlib

# hashlib

# json
